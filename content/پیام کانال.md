🌻[صفّ۸] ص۵۵۲ - یُریدُونَ لِیُطْفِؤُا نُورَ اللَّهِ بِأَفْواهِهِمْ وَ اللَّهُ مُتِمُّ نُورِهِ وَ لَوْ کَرِهَ الْکافِرُونَ
می‌خواهند نور خدا را با دهان خود خاموش کنند و حال آنکه خدا -گر چه کافران را ناخوش افتد- نور خود را کامل خواهد گردانید.
🌻[صفّ۹] ص۵۵۲ - هُوَ الَّذی أَرْسَلَ رَسُولَهُ بِالْهُدى‏ وَ دینِ الْحَقِّ لِیُظْهِرَهُ عَلَى الدِّینِ کُلِّهِ وَ لَوْ کَرِهَ الْمُشْرِکُونَ
اوست کسی که فرستاده خود را با اَلهدی و دین حقّ روانه کرد، تا آن را بر هر چه دین است ظاهر گرداند، هر چند مشرکان را ناخوش آید.
🌻[صفّ۱۰] ص۵۵۲ - يَا أَيُّهَا الَّذِينَ آمَنُوا هَلْ أَدُلُّكُمْ عَلَىٰ تِجَارَةٍ تُنْجِيكُمْ مِنْ عَذَابٍ أَلِيمٍ 
ای اهل ایمان! آیا شما را به تجارتی راهنمایی کنم که شما را از عذابی دردناک نجات می دهد؟
🌻[صفّ۱۱] ص۵۵۲ - تُؤْمِنُونَ بِاللَّهِ وَرَسُولِهِ وَتُجَاهِدُونَ فِي سَبِيلِ اللَّهِ بِأَمْوَالِكُمْ وَأَنْفُسِكُمْ ۚ ذَٰلِكُمْ خَيْرٌ لَكُمْ إِنْ كُنْتُمْ تَعْلَمُونَ
به خدا و پیامبرش ایمان آورید، و با اموال و جان هایتان در راه خدا جهاد کنید؛ این [ایمان و جهاد] اگر [به منافع فراگیر و همیشگی آن] معرفت و آگاهی داشتید، برای شما [از هر چیزی] بهتر است
🌻[صفّ۱۴] ص۵۵۲ - یا أَیُّهَا الَّذینَ آمَنُوا کُونُوا أَنْصارَ اللَّهِ کَما قالَ عیسَى ابْنُ مَرْیَمَ لِلْحَوارِیِّینَ مَنْ أَنْصاری إِلَى اللَّهِ قالَ الْحَوارِیُّونَ نَحْنُ أَنْصارُ اللَّهِ فَآمَنَتْ طائِفَةٌ مِنْ بَنی‏ إِسْرائیلَ وَ کَفَرَتْ طائِفَةٌ فَأَیَّدْنَا الَّذینَ آمَنُوا عَلى‏ عَدُوِّهِمْ فَأَصْبَحُوا ظاهِرینَ
ای کسانی که ایمان آورده‌اید، یاران خدا باشید، همان گونه که عیسی‌بن‌مریم به حواریون گفت: «یاران من در راه خدا چه کسانی‌اند؟» حواریون گفتند: «ما یاران خداییم.» پس طایفه‌ای کفر ورزیدند، و [ما] کسانی را که ایمان آوردند، بر دشمنانشان یاری کردیم تا چیره شدند.

چگونه با امامی که نمی بینیم همراهی کنیم؟

برای پاسخ به این سؤال ابتدا می بایست پاسخ سؤالات زیر را «در قرآن» بیابیم:
فاعل حوادث و صحنه ها به اذن االله امام علیه السلام اند، ایشان دقیقاً کیست یا چیست؟ چه برنامه ای از سوی خدا دارند؟ این برنامه از ارتفاعات بسوی عمق چگونه نازل می شود؟ برنامه ی ایشان در این مقطع از زمانه ی ما، چیست؟ امام (ع) چگونه در حال اِعمال برنامه ی شان هستند؟ مردم چه سهم یا نقشی در این برنامه دارند؟ چگونه باید برای همراهی با فرمانده تربیت شد؟ 
امام (ع) سرباز می خواهد، سرباز باید تربیت شود.

😳⁉️ ➖آیا باید حتما برای این تربیت و همپایی با امام علیه السلام، فقط با قرآن و کلام اهل بیت همراه شد؟

اگر از هر جایی هر پاسخی شنیدیم که نیاز و اتصال ما را به امام (ع) کم رنگ می کند و ما را بیشتر به تلاش خود، وعده می دهد و وصل می کند، بدانید که غیر کلام قرآن و خواست خداست. چرا که تمام قرآن، پیوسته ما را به سوی امام و اتصال به امام دعوت می کند. همه چیزمان را باید از او در قرآن و اهل بیت (ع) بگیریم. غیر این، آغاز تدریجیِ الضَّالِّینَ شدن 🧌 است و بس! 
إِنَّ هذَا الْقُرْآنَ یَهْدی لِلَّتی‏ هِیَ أَقْوَمُ  [إسراء] ص۲۸۳ 
قطعا این قرآن [است که] به چیزی که قوام‌دارتر، قائم تر، استوارتر است هدایت می کند.
در خصوص این آیه امام صادق (ع) فرمودند: وَ الْإِمَامُ یَهْدِی إِلَی الْقُرْآنِ وَ الْقُرْآنُ یَهْدِی إِلَی الْإِمَامِ وَ ذَلِکَ قَوْلُ اللَّهِ عزّوجلّ إِنَّ هذَا الْقُرْآنَ یَهْدِی لِلَّتِی هِیَ أَقْوَمُ.
و امام به‌سوی قرآن هدایت می‌کند و قرآن به‌سوی امام هدایت می‌کند و این معنی کلام خداست که می‌فرماید: إِنَّ هَذَا الْقُرْآنَ یِهْدِی لِلَّتِی هِیَ أَقْوَمُ.

3 | چگونه بسوی یک موضوع مجهول در قرآن پیش برویم؟  [صفّ1] ص۵۵۲

ابتدا ویژگی ها و مشخصاتی پیرامون آن موضوع در همان آیه ای که آن کلمه وجود دارد، پیدا کن
پس از آن، همین جستجو و جمع آوری اطلاعات را در آیات قبل و بعد آن (سیاق آیه) انجام بده
سپس در ادامه ی سوره یا حتی در دیگر سوره ها همان جمع آوری را انجام بده
اطلاعات و مشخصات جمع آوری شده را دسته بندی و اولویت بندی کن
جمع بندی را جای ترجمه ی آن واژه قرار بده  آیه را ترجمه و فهم کن
اگر درست و صحیح به درجه ای بالاتر از فهم آن کلمه، به لطف خدا رسانده شده ای، می بایست در تصویر نهاییِ پازل مجموع آیات، درست عمل کند قوی تر به نظر خدا در سیاق آیه یا سوره نزدیک شده باشی. (نظر نهایی از یک کلمه یا آیه را فقط خداوند و اهل بیت علیهم السلام می دانند ما فقط سعی می کنیم به توفیق الهی با عبور از پلکان علما و بزرگان دین به نظر خداوند، نزدیک تر شویم)

4 | «فی‏ سَبیلِ اللَّهِ» یعنی چی؟ اجازه نداریم به هر موضوعی نسبت بدهیم.  [صفّ4] ص۵۵۲

ــــــــــــــــــــــــــــــــــ
🌻 [صفّ۱] ص۵۵۱ - کسی چیزی را در آسمان ها و زمین جریان داد (تسبیح داد): سَبَّحَ لِلَّهِ ما فِی السَّماواتِ وَ ما فِی الْأَرْضِ وَ هُوَ الْعَزیزُ الْحَکیمُ
آنچه در آسمانها و آنچه در زمین است به تسبیح خدا می‌پردازند، و اوست ارجمند حکیم.
🌻[صفّ۲] ص۵۵۱ - خداوند در آیات 2 و 3 از کسانی به خاطر رفتارشان ناراحت است:  یا أَیُّهَا الَّذینَ آمَنُوا لِمَ تَقُولُونَ ما لا تَفْعَلُونَ
ای کسانی که ایمان آورده‌اید، چرا چیزی می‌گویید که انجام نمی‌دهید؟
🌻[صفّ۳] ص۵۵۱ - کَبُرَ مَقْتاً عِنْدَ اللَّهِ أَنْ تَقُولُوا ما لا تَفْعَلُونَ
نزد خدا سخت ناپسند است که چیزی را بگویید و انجام ندهید.
🌻[صفّ۴] ص۵۵۱ -خداوند کسانی را دوست دارد. اینان در راه برپایی ظهور همپای امام ع چون کوهی آهنین قِتال یا جنگ می کنند:  إِنَّ اللَّهَ یُحِبُّ الَّذینَ یُقاتِلُونَ فی‏ سَبیلِهِ صَفًّا کَأَنَّهُمْ بُنْیانٌ مَرْصُوصٌ
در حقیقت، خدا دوست دارد کسانی را که در راه او صف در صف، چنانکه گویی بنایی ریخته شده از سرب‌اند، جهاد می‌کنند.
🌻  [صفّ۷] ص۵۵۲ - همان کسی که جریان را تسبیح می دهد، دارد به اسلام دعوت می کند:  وَ مَنْ أَظْلَمُ مِمَّنِ افْتَرى‏ عَلَى اللَّهِ الْکَذِبَ وَ هُوَ یُدْعى‏ إِلَى الْإِسْلامِ وَ اللَّهُ لا یَهْدِی الْقَوْمَ الظَّالِمینَ
و چه کسی ستمگرتر از آن کس است که با وجود آنکه به سوی اسلام فراخوانده می شود، بر خدا دروغ می‌بندد؟ و خدا مردم ستمگر را راه نمی‌نماید.
🌻  [صفّ۸] ص۵۵۲ -کسانی می خواهند با توان های غیر مادی و نیز مادی این جریان نورانی خدا را (نورالله) خاموش کنند خداوند نمی گذارد:  یُریدُونَ لِیُطْفِؤُا نُورَ اللَّهِ بِأَفْواهِهِمْ وَ اللَّهُ مُتِمُّ نُورِهِ وَ لَوْ کَرِهَ الْکافِرُونَ
می‌خواهند نور خدا را با دهان خود خاموش کنند و حال آنکه خدا -گر چه کافران را ناخوش افتد- نور خود را کامل خواهد گردانید.
🌻[صفّ۹] ص۵۵۲ - خداوند نورالله اش را به صورت مرسوله ای به رسول اکرم (ص) رسالت می دهد تا با الهدی (امام زمان عج) و دین حق، ظهور را برپا کنند هُوَ الَّذی أَرْسَلَ رَسُولَهُ بِالْهُدى‏ وَ دینِ الْحَقِّ لِیُظْهِرَهُ عَلَى الدِّینِ کُلِّهِ وَ لَوْ کَرِهَ الْمُشْرِکُونَ
اوست کسی که فرستاده خود را با هدایت و آیین درست روانه کرد، تا آن را بر هر چه دین است فائق گرداند، هر چند مشرکان را ناخوش آید.

6 | مفهوم تسبیح چیست؟ الآن چه ماجرایی در ایران در حال تسبیح است؟  [صفّ1-9] ص۵۵۲

تسبیح: 
لغات همچون یک مکعب دارای «وجوه معنایی» هستند.
در ارتفاعات، یا یکی از مرتفع ترین وجوه معناییِ تسبیح در «این زمانه»:
جریان دادن است، شناور بودن، جریانی که همراه با آن، موانع، نقایص و حُجُب برطرف می شود.
مثل شناگری که حین جریان داشتن در آب، آب را نیز از سر راه خود تنزیه می کند، پس می زند.
هر قدر نورِ ظهور، بیشتر و بیشتر شود، تحت تاثیرِ آن، باید به دنبال معانی و مفاهیم مرتفع تری از یک واژه ی قرآنی باشیم، 
چراکه:
«راکد ماندن بر فهم از قرآن جفاست!»

6 | مفهوم تسبیح چیست؟ اکنون چه ماجرایی در ایران در حال تسبیح است؟  [صفّ1-9] ص۵۵۲

تسبیح یا ازدیاد جریانِ:
 جریان ما فِی السَّماواتِ وَ ما فِی الْأَرْضِ = جریان سَبیلِهِ = جریان اسْمُهُ أَحْمَدُ = جریان یُدْعى‏ إِلَى الْإِسْلامِ (دعوت به اسلام) =  جریان نُورَ اللَّهِ = جریان رسالت رسول الله (ص) = جریان یُظْهِرَهُ عَلَى الدِّینِ کُلِّهِ (ظهور)، همراه با پس زدن موانع، نقایص، مقاومت ها و حُجُب بوسیله ی دو اسم الْعَزیزُ الْحَکیمُ، در ایران است

خلاصه: ماجرای ازدیاد نوراللهِ امام عج یا همان جریان ظهور(مهدویت) در حال تسبیح است (تام شدن همراه با پس زدن تمامی موانع، به چه صورت؟ ظاهر شدن بر تمامی ادیان)

 لغت‌نامه‌های عربی:
 لسان‌العرب (ابن‌منظور)، الصحاح (جوهري)، القاموس المحيط (فيروزآبادی) – «سبح» یعنی شنا کردن، حرکت در آب یا هوا.

دهخدا، معین، عمید – «تسبیح» و «سباحة» به معنای شنا، شناور، شناگر، جریان داشتن. 
 لغت‌نامه های فارسی معاصر:
 فرهنگ معاصر عربی-فارسی (آذرتاش آذرنوش) تأیید شناور بودن.

🟢 چگونه از نماز استفاده کنیم؟🙇‍♂️

⁉️#سؤال: من از کلیّت و جزئیاتِ نمازم چگونه برای رشدم استفاده کنم؟ 

✍️... ببینید، ما باید از کلیّت نماز و نیز از جمله جمله نماز و از بخش‌های مختلف آن، برای رشد کردن استفاده کنیم؛ مثلاً در جزئیات نماز مثال بزنیم، #ابلیسِملعون بر آدم (ع) سجده نکرد، خداوند هم طردش کرد؛ اما ملائکه بر ایشان سجده کردند و خدا آن‌ها را در جایگاهِ خودشان #حفظ کرد؛ این یک شگرد در استفاده گرفتن از نماز برای رشد است.

🔹پس سجده یکی از ارکانِ مهم در نماز است که به #رشدِ من ربط دارد. من می‌خواهم رشد کنم و #ارتفاع بگیرم؛ لذا باید در پیشگاه خدا سجده کنم؛ بدونِ سجده، انسان از درگاهِ خداوندی #طرد می‌شود و با سجده، انسان در جایگاهِ خوبی قرار می‌گیرد که تازه این در حدّ ملائک است؛ و برخوردِ خداوند برای سجده‌ی انسان، از این هم شأنش بالاتر است.

✅ما باید از #ظرفیتِ نمازِ اولِ وقت برای رشدمان استفاده کنیم؛ همچنین از ظرفیتِ سجده و ظرفیتِ رکوع نیز باید استفاده کنیم، بلکه از ظرفیتِ تک‌تکِ جملاتِ مختلف نماز باید در جهتِ رشد استفاده کنیم! اما رشد به سوی چه چیزی؟ رشد باید جهت و مقصد داشته باشد. صرفاً رشدِ به سوی خدا، تیرِ هواییِ بدونِ هدف است، جهت دارد؛ اما مقصد ندارد؛ چون خداوند در جایی مستقر نیست که من با #نمازم به آنجا بروم. پس باید چه کرد؟ چه مقصدی را برای رشد در نماز باید به عنوانِ #هدف در نظر گرفت؟

🔸پاسخ در #جامعه‌ی_کبیره است که: «مَنْ أَرَادَ اللَّهَ بَدَأَ بِكُمْ وَ مَنْ وَحَّدَهُ قَبِلَ عَنْكُمْ وَ مَنْ قَصَدَهُ تَوَجَّهَ بِكُمْ» کسی که خدا را #اراده کند، از شما شروع می‌کند؛ و کسی که #واجد_توحید شود، به شما #اقبال می‌کند؛ و کسی که مقصدش خدا باشد؛ متوجه شما می‌‌شود. پس ببینید، برای #رشد به سوی خدا، باید رو به سوی اهل‌بیت (ع) باشیم؛ رشد در ولایتِ اهل‌بیت (ع)، توجه به امامتِ ایشان و رفتن به‌سوی محضرِ مبارکشان از توحید جدا نیست، لذا برای اینکه ما در بیتُ الله باشیم، باید در بیتِ اهل‌بیت (ع) یا مثلاً در خانه‌ی مادر مهربان (س) وارد شویم، برای اینکه متوجه خدا باشیم؛ باید متوجه ایشان باشیم، در محضرِ ایشان باشیم و اینگونه در #نمازمان، خدا، خداییِ خدا و عبودیتش را دنبال کنیم تا ما را رشد بدهد.

🕋 #حج، ماکتی از #مهدویت!

✍️... اميدوارم که خداوند در #ظهور مولايمان امام زمان(عج) بسيار تعجيل بفرمايد. اميدوارم که خداوند #معرفت امام زمان(عج) را بیش از پیش به همه‌ی ما عطا کند. #دورانِ_غيبت، #دورانِ_آخرالزمان و #دورانِ_ظهور به حسبِ ماکت اگر بخواهیم به آن بنگریم؛ یعنی به حسبِ «ماجرای حج» اينطور می‌شود که: 
دوران #غيبت يعنی دورانِ #معرفت که مابه‌ازاء آن، «عرفات» در مناسکِ #حج است. 
دوران آخرالزمان و قرار گرفتنِ انسان در صحنه‌ها و آیاتِ #آخرالزمانی که به نوعی دورانِ #شعورِ بشریت نسبت به #مهدویت است، ما به ازاءش در حج، می‌شود «مشعر» 
و دورانِ ظهورِ مهدی(عج) که دورانِ اصلی #جنگ‌های جبهه‌ی حق علیه #ابلیس و دیگر #شیاطین است، ما به ازاءش در حج می‌شود «منا».

عرفات= غیبت
مشعر= آخرالزمان
مِنا= ظهور

📿حاجيان، در یک کلاسِ تمرینی به نامِ مناسکِ حج، بعد از اینکه از لباس و هویت دنیا خارج شده و #مُحرِم شدند؛ دسته‌جمعی در موقعیتِ #عرفات برده می‌شوند، آنجا کمی فرصت دارند تا #معرفت‌شان را تقويت کنند. سپس به یک وادی در آستانه‌ی #منا برده می‌شوند که به آنجا #مشعر گفته می‌شود، تا در آنجا به شعور بپردازند؛ شعور یعنی داشتن حس، داشتن فهم و احساس مناسبِ یک صحنه، لذا حاجی در مشعر باید #شعور مناسب نسبت به منا که صحنه‌ی مواجهه با #ابلیس هست را؛ واجد شود.

🔔پس شعور يعنی چه؟ يعنی به دنبال #معرفتِ_قلبی در عرفات، حال «آماده شدنِ ظاهری» در مشعر کسب شود. وقتی خداوند در #قرآن مثلاً در آیه‌ی ۱۰۷ یوسف می‌فرمايد: «وَ هُمْ لايَشْعُرُونَ»، يعنی اين‌ها خودشان را برای مواجهه با این صحنه‌ها آماده نکرده‌اند. تصور کنيد که #سربازان دارند در مسيری حرکت می‌کنند، اصلاً برای جنگ آمده‌اند، این يعنی معرفتِ لازم را دارند. ناگهان در #کمينِ دشمن می‌افتند، يعنی برای کمينِ دشمن اصلاً آماده نبودند.در این صحنه می‌شود گفت که این سربازان اينجا معرفت را داشتند، اما شعور را نداشتند. خودشان را برای در افتادن با دشمن آماده نکرده بودند، لذا در کمينِ دشمن افتادند.

💥#آخرالزمان، دورانِ ما به ازاء سرزمين #مشعر يعنی محلی برای کسبِ #شعور است. 
الآن که ما در اواخر دوران غیبت و آستانه ی دوران آخرالزمانِ حوادث هستیم، بیاییم با سرعت در کسبِ #معرفتِ_مهدوی، معرفتمان را نسبت به مقولات مختلف از جمله نسبت به برنامه ی خدا در این مقطع، امام شناسی و تربیت های لازمهٔ آن، ارتقاء دهیم.
